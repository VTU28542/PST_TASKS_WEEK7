import java.io.*;
import java.util.*;

public class Solution {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();

            if (x % 2 == 0) {
                System.out.println(x + 2);
            } else {
                System.out.println(x);
            }
        }

        sc.close();
    }
}
